### 5.6 综合示例：用户信息登记表（约 80 行）

功能：姓名/密码/性别（单选互斥）/城市下拉/爱好多选/自我介绍 + 进度条反映必填完成度 + 提交汇总到摘要 Label。Qt5/Qt6 均可编译。

```cpp
// main.cpp —— 类写在 .cpp 且带 Q_OBJECT，末尾必须 #include "main.moc"
#include <QApplication>
#include <QWidget>
#include <QLabel>
#include <QLineEdit>
#include <QComboBox>
#include <QCheckBox>
#include <QRadioButton>
#include <QButtonGroup>
#include <QPlainTextEdit>
#include <QPushButton>
#include <QProgressBar>
#include <QGroupBox>
#include <QFormLayout>
#include <QVBoxLayout>
#include <QHBoxLayout>
#include <QList>
#include <QDate>

class RegisterForm : public QWidget {
    Q_OBJECT
public:
    explicit RegisterForm(QWidget *parent = nullptr) : QWidget(parent) {
        setWindowTitle("用户信息登记表");
        m_name = new QLineEdit;  m_name->setPlaceholderText("2-20 个字符");
        m_pass = new QLineEdit;  m_pass->setEchoMode(QLineEdit::Password);
        m_pass->setPlaceholderText("至少 6 位");
        m_city = new QComboBox;  m_city->addItems({"北京", "上海", "深圳", "杭州"});
        m_intro = new QPlainTextEdit; m_intro->setPlaceholderText("一句话介绍自己");

        auto *genderBox = new QGroupBox("性别");
        auto *genderLay = new QHBoxLayout(genderBox);
        m_male = new QRadioButton("男");
        auto *female = new QRadioButton("女");
        m_male->setChecked(true);
        auto *genderGroup = new QButtonGroup(this);   // 同组自动互斥
        genderGroup->addButton(m_male);
        genderGroup->addButton(female);
        genderLay->addWidget(m_male);  genderLay->addWidget(female);

        auto *hobbyBox = new QGroupBox("爱好（可多选）");
        auto *hobbyLay = new QHBoxLayout(hobbyBox);
        for (const char *h : {"C++", "Qt", "游戏", "运动"}) {
            auto *cb = new QCheckBox(h);
            m_hobbies.append(cb);
            hobbyLay->addWidget(cb);
        }

        auto *form = new QFormLayout;
        form->addRow("姓名：", m_name);
        form->addRow("密码：", m_pass);
        form->addRow("城市：", m_city);
        form->addRow(genderBox);            // 整行控件：addRow(QWidget*)
        form->addRow(hobbyBox);
        form->addRow("自我介绍：", m_intro);

        m_bar = new QProgressBar;  m_bar->setRange(0, 4);
        m_summary = new QLabel("等待提交…");
        m_summary->setWordWrap(true);
        m_summary->setAlignment(Qt::AlignTop | Qt::AlignLeft);
        auto *submit = new QPushButton("提交");

        auto *right = new QVBoxLayout;
        right->addWidget(m_bar);
        right->addWidget(m_summary, 1);     // 摘要占剩余空间
        right->addWidget(submit);

        auto *main = new QHBoxLayout(this);
        main->addLayout(form, 3);
        main->addLayout(right, 2);

        // textEdited 只响应用户输入；textChanged 连程序 setText 也触发
        connect(m_name,  &QLineEdit::textEdited, this, &RegisterForm::refreshProgress);
        connect(m_pass,  &QLineEdit::textEdited, this, &RegisterForm::refreshProgress);
        connect(m_intro, &QPlainTextEdit::textChanged, this, &RegisterForm::refreshProgress);
        connect(submit,  &QPushButton::clicked, this, &RegisterForm::onSubmit);
    }
private slots:
    void refreshProgress() {
        int done = 1;                               // 性别默认已选
        if (!m_name->text().isEmpty())         ++done;
        if (m_pass->text().size() >= 6)        ++done;
        if (!m_intro->toPlainText().isEmpty()) ++done;
        m_bar->setValue(done);
    }
    void onSubmit() {
        QStringList hobbies;
        for (QCheckBox *cb : m_hobbies)
            if (cb->isChecked()) hobbies << cb->text();
        m_summary->setText(QString(
            "姓名：%1\n性别：%2\n城市：%3\n爱好：%4\n简介：%5\n登记日期：%6")
            .arg(m_name->text(),
                 m_male->isChecked() ? "男" : "女",
                 m_city->currentText(),
                 hobbies.isEmpty() ? "无" : hobbies.join("、"),
                 m_intro->toPlainText().isEmpty() ? "未填写" : m_intro->toPlainText(),
                 QDate::currentDate().toString("yyyy-MM-dd")));
    }
private:
    QLineEdit *m_name, *m_pass;
    QComboBox *m_city;
    QRadioButton *m_male;
    QList<QCheckBox*> m_hobbies;
    QPlainTextEdit *m_intro;
    QProgressBar *m_bar;
    QLabel *m_summary;
};

int main(int argc, char *argv[]) {
    QApplication a(argc, argv);
    RegisterForm w;
    w.resize(760, 420);
    w.show();
    return a.exec();
}

#include "main.moc"   // Q_OBJECT 类在 .cpp 中时必需（qmake 与 AUTOMOC 都认）
```

## 6. 布局管理

### 6.1 为什么需要布局

手动 move/resize 的绝对定位：窗口拉伸后控件原地不动，字体/DPI/翻译切换后互相遮挡。布局管理器按规则分配空间，控件随窗口自动伸缩。

| | 绝对定位 | 布局管理 |
|---|---|---|
| 窗口缩放 | 不跟随 | 自动重排 |
| 字体/翻译变化 | 易重叠错位 | 自适应 |
| 控件增删 | 手工挪位 | 自动腾位 |
| 适用 | 游戏界面、悬浮球等特殊场景 | 99% 的表单/工具窗口 |

### 6.2 四大布局类

| 布局 | 排列 | 核心函数 | 典型场景 |
|---|---|---|---|
| QHBoxLayout | 水平一行 | addWidget(w, stretch) | 按钮行 |
| QVBoxLayout | 垂直一列 | addWidget(w, stretch) | 表单骨架 |
| QGridLayout | 二维网格 | addWidget(w, row, col, rowSpan, colSpan, align)；setColumnStretch(col, f) / setRowStretch(row, f) | 计算器、设置页 |
| QFormLayout | 标签-字段两列 | addRow("用户名：", edit) / addRow(label, field) / addRow(QWidget*) | 登录、属性面板 |

```cpp
auto *grid = new QGridLayout;
grid->addWidget(nameEdit, 0, 0);   grid->addWidget(passEdit, 0, 1);
grid->addWidget(rememberBox, 1, 0, 1, 2);     // 跨 1 行 2 列
grid->setColumnStretch(1, 1);                 // 第 2 列吃掉额外宽度

auto *form = new QFormLayout;
form->addRow(tr("服务器："), hostEdit);
form->addRow(tr("端口："),   portSpin);
```

### 6.3 关键方法表

| 方法 | 说明 | 示例 |
|---|---|---|
| addWidget(w, stretch, alignment) | 加入控件，可带伸展因子与对齐 | `lay->addWidget(list, 1);` |
| addLayout(child) | 布局嵌套 | |
| addSpacing(int) | 固定空白（占位不伸缩） | `lay->addSpacing(20);` |
| addStretch(int stretch=0) | **弹性弹簧**：吃掉全部剩余空间【高频】 | `btns->addStretch(1);` |
| setSpacing(int) / setContentsMargins(l,t,r,b) | 控件间距 / 布局外边距 | |
| setStretchFactor(w 或 index, f) | 事后调比例 | |
| insertWidget / removeWidget / removeItem | 插入 / 移出（不删除） | |

### 6.4 把布局挂到界面上

```cpp
auto *lay = new QVBoxLayout;
lay->addWidget(...);
w->setLayout(lay);            // 挂载；等价捷径：new QVBoxLayout(w)
```

要点：加入布局的控件会被**自动 reparent** 到布局最终挂载的 widget（new 时没传 parent 也行）；一个 widget 只能挂一个布局——**重复 setLayout 不会替换**，只警告，须先 `delete w->layout()`；布局嵌套思想：先拼最小单元再逐层组装。

### 6.5 尺寸控制：sizePolicy 七种策略

| 策略 | 含义 |
|---|---|
| Fixed | 尺寸固定为 sizeHint，不可伸缩 |
| Minimum | sizeHint 是最小值，只能放大 |
| Maximum | sizeHint 是最大值，只能缩小 |
| Preferred（多数控件默认） | sizeHint 是理想值，可伸可缩 |
| MinimumExpanding | sizeHint 是最小值，并主动争抢剩余空间 |
| Expanding | sizeHint 是理想值，可伸可缩，且**优先分得额外空间**（与 Preferred 同行时多余空间归它） |
| Ignored | 无视 sizeHint，尽力伸缩 |

```cpp
btn->setSizePolicy(QSizePolicy::Fixed, QSizePolicy::Fixed);
edit->setSizePolicy(QSizePolicy::Expanding, QSizePolicy::Preferred);
btn->setFixedSize(96, 32);
list->setMinimumWidth(180);
w->adjustSize();                       // 按 sizeHint 收紧窗口
w->layout()->setSizeConstraint(QLayout::SetFixedSize);   // 窗口锁死为布局理想大小
```

### 6.6 经典示例：三按钮底部右对齐（addStretch 用法）

```cpp
auto *root = new QVBoxLayout(this);
root->addWidget(textEdit, 1);          // 编辑器占主要空间

auto *btns = new QHBoxLayout;
btns->addStretch(1);                   // 左侧弹簧：把按钮推向右边【高频】
btns->addWidget(helpBtn);
btns->addWidget(cancelBtn);
btns->addWidget(okBtn);
root->addLayout(btns);
root->addSpacing(8);
```

```text
┌───────────────────────────────────────────┐
│ QTextEdit（stretch=1，占主要高度）          │
├───────────────────────────────────────────┤
│ ←── addStretch（弹簧）──→ [帮助][取消][确定] │
└───────────────────────────────────────────┘
```


### 7.5 综合示例：迷你记事本（约 100 行）

菜单【文件：新建/打开/保存/退出】【编辑：撤销/剪切/复制/粘贴】【帮助：关于】，QTextEdit 中央区，QFileDialog 读写文件，未保存关闭询问（closeEvent），关于用 QMessageBox。Qt5/Qt6 均可编译。

```cpp
// ================= mainwindow.h =================
#pragma once
#include <QMainWindow>

class QTextEdit;

class MainWindow : public QMainWindow {
    Q_OBJECT
public:
    explicit MainWindow(QWidget *parent = nullptr);
protected:
    void closeEvent(QCloseEvent *event) override;   // 关闭前询问保存
private slots:
    void newFile();
    void openFile();
    bool saveFile();
    void about();
private:
    void createActions();
    bool maybeSave();
    void setCurrentFile(const QString &fileName);
    QTextEdit *m_edit;
    QString m_currentFile;
    bool m_dirty = false;
};

// ================= mainwindow.cpp =================
#include "mainwindow.h"
#include <QTextEdit>
#include <QMenuBar>
#include <QMenu>
#include <QToolBar>
#include <QStatusBar>
#include <QAction>        // Qt5 在 QtWidgets，Qt6 移到 QtGui（include 不变）
#include <QFileDialog>
#include <QMessageBox>
#include <QFileInfo>
#include <QFile>
#include <QTextStream>

MainWindow::MainWindow(QWidget *parent) : QMainWindow(parent) {
    setWindowTitle(tr("未命名[*] - 迷你记事本"));  // [*] 配合 setWindowModified 显示星号
    resize(800, 600);
    m_edit = new QTextEdit(this);
    setCentralWidget(m_edit);                    // 中央控件：有且只有一个
    createActions();
    statusBar()->showMessage(tr("就绪"), 3000);
    connect(m_edit, &QTextEdit::textChanged, this, [this] {  // 变化→置脏
        m_dirty = true;
        setWindowModified(true);
    });
}

void MainWindow::createActions() {
    QAction *newAct = new QAction(tr("&新建"), this);
    newAct->setShortcut(QKeySequence::New);
    connect(newAct, &QAction::triggered, this, &MainWindow::newFile);

    QAction *openAct = new QAction(tr("&打开..."), this);
    openAct->setShortcut(QKeySequence::Open);
    connect(openAct, &QAction::triggered, this, &MainWindow::openFile);

    QAction *saveAct = new QAction(tr("&保存"), this);
    saveAct->setShortcut(QKeySequence::Save);
    connect(saveAct, &QAction::triggered, this, &MainWindow::saveFile);

    QAction *exitAct = new QAction(tr("退出"), this);
    exitAct->setShortcut(tr("Ctrl+Q"));
    // 走 close() 而非直接退出，才能触发 closeEvent 的保存询问
    connect(exitAct, &QAction::triggered, this, &QWidget::close);

    // 编辑类动作：直接复用 QTextEdit 自带槽
    QAction *undoAct = new QAction(tr("撤销"), this);
    undoAct->setShortcut(QKeySequence::Undo);
    connect(undoAct, &QAction::triggered, m_edit, &QTextEdit::undo);
    QAction *cutAct = new QAction(tr("剪切"), this);
    cutAct->setShortcut(QKeySequence::Cut);
    connect(cutAct, &QAction::triggered, m_edit, &QTextEdit::cut);
    QAction *copyAct = new QAction(tr("复制"), this);
    connect(copyAct, &QAction::triggered, m_edit, &QTextEdit::copy);
    QAction *pasteAct = new QAction(tr("粘贴"), this);
    connect(pasteAct, &QAction::triggered, m_edit, &QTextEdit::paste);

    QAction *aboutAct = new QAction(tr("关于"), this);
    connect(aboutAct, &QAction::triggered, this, &MainWindow::about);

    QMenu *fileMenu = menuBar()->addMenu(tr("&文件"));
    fileMenu->addAction(newAct);
    fileMenu->addAction(openAct);
    fileMenu->addAction(saveAct);
    fileMenu->addSeparator();
    fileMenu->addAction(exitAct);

    QMenu *editMenu = menuBar()->addMenu(tr("&编辑"));
    editMenu->addAction(undoAct);
    editMenu->addSeparator();
    editMenu->addAction(cutAct);
    editMenu->addAction(copyAct);
    editMenu->addAction(pasteAct);

    menuBar()->addMenu(tr("&帮助"))->addAction(aboutAct);

    QToolBar *tool = addToolBar(tr("主工具栏"));   // 工具栏复用同一批 QAction
    tool->setMovable(false);
    tool->addAction(newAct);
    tool->addAction(openAct);
    tool->addAction(saveAct);
    tool->addSeparator();
    tool->addAction(undoAct);
}

void MainWindow::newFile() {
    if (!maybeSave()) return;
    m_edit->clear();
    setCurrentFile(QString());
}

void MainWindow::openFile() {
    if (!maybeSave()) return;
    // 过滤器语法：组内多扩展名空格分隔，组间用 ";;"【高频】
    QString fileName = QFileDialog::getOpenFileName(this, tr("打开文件"), QString(),
        tr("文本文件 (*.txt);;C++ 源码 (*.cpp *.h);;所有文件 (*)"));
    if (fileName.isEmpty()) return;               // 空串=用户取消，必须判断
    QFile file(fileName);
    if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
        QMessageBox::warning(this, tr("错误"), tr("无法打开文件：") + file.errorString());
        return;
    }
    QTextStream in(&file);
    // Qt5：in.setCodec("UTF-8");  Qt6：setCodec 已移除，默认即 UTF-8
    m_edit->setPlainText(in.readAll());
    setCurrentFile(fileName);
}

bool MainWindow::saveFile() {
    if (m_currentFile.isEmpty()) {
        QString fileName = QFileDialog::getSaveFileName(this, tr("保存文件"), QString(),
            tr("文本文件 (*.txt);;所有文件 (*)"));
        if (fileName.isEmpty()) return false;
        m_currentFile = fileName;
    }
    QFile file(m_currentFile);
    if (!file.open(QIODevice::WriteOnly | QIODevice::Text | QIODevice::Truncate)) {
        QMessageBox::warning(this, tr("错误"), tr("无法写入文件：") + file.errorString());
        return false;
    }
    QTextStream out(&file);
    out << m_edit->toPlainText();
    m_dirty = false;
    setWindowModified(false);
    statusBar()->showMessage(tr("已保存 %1").arg(m_currentFile), 3000);
    return true;
}

bool MainWindow::maybeSave() {
    if (!m_dirty) return true;
    auto ret = QMessageBox::warning(this, tr("迷你记事本"), tr("文档已修改，是否保存？"),
        QMessageBox::Save | QMessageBox::Discard | QMessageBox::Cancel);
    if (ret == QMessageBox::Save)   return saveFile();
    if (ret == QMessageBox::Cancel) return false;
    return true;                                  // Discard：放弃修改
}

void MainWindow::closeEvent(QCloseEvent *event) {
    if (maybeSave()) event->accept();             // 放行关闭
    else             event->ignore();             // 拦截
}

void MainWindow::setCurrentFile(const QString &fileName) {
    m_currentFile = fileName;
    m_dirty = false;
    setWindowModified(false);
    setWindowTitle(tr("%1[*] - 迷你记事本").arg(
        fileName.isEmpty() ? tr("未命名") : QFileInfo(fileName).fileName()));
}

void MainWindow::about() {
    QMessageBox::about(this, tr("关于"),
        tr("<b>迷你记事本</b><br>Qt Widgets 综合示例<br>Qt 版本：%1").arg(QT_VERSION_STR));
}

// ================= main.cpp =================
#include <QApplication>
#include "mainwindow.h"

int main(int argc, char *argv[]) {
    QApplication a(argc, argv);
    MainWindow w;
    w.show();
    return a.exec();
}
```

### 8.1 模态 vs 非模态【高频】

| | 模态 | 非模态 |
|---|---|---|
| 显示方式 | `dlg.exec()` | `dlg.show()` |
| 行为 | 阻塞父窗口输入直到关闭返回 | 立即返回，可同时操作主窗口 |
| 返回值 | int：QDialog::Accepted(1)/Rejected(0) | 无（结果经信号/取数函数获取） |
| 创建位置 | 可栈上（exec 返回时仍存活） | **必须堆上**，否则函数返回即析构【高频】 |
| 防泄漏 | —— | `setAttribute(Qt::WA_DeleteOnClose)`：关闭自动销毁 |

```cpp
// 模态：栈上 + exec
SettingsDialog dlg(this);
if (dlg.exec() == QDialog::Accepted) applySettings(dlg.values());

// 非模态：堆上 + 关闭自删（有 parent：主窗口析构级联清理）
auto *find = new FindDialog(this);
find->setAttribute(Qt::WA_DeleteOnClose);
find->show();
```

### 8.6 自定义对话框（Qt Designer / .ui 方式）

Designer 拖出的界面存为 `.ui`，由 uic 生成 `ui_dialog.h`。C++ 侧常用**多继承**（也可用组合成员 `Ui::Dialog ui;`）：

```cpp
// mydialog.h —— 假设 ui 文件顶层对象名为 Dialog
#pragma once
#include <QDialog>
#include "ui_dialog.h"           // uic 生成；qmake 需 FORMS += dialog.ui

class MyDialog : public QDialog, public Ui::Dialog {
    Q_OBJECT
public:
    explicit MyDialog(QWidget *parent = nullptr) : QDialog(parent) {
        setupUi(this);           // 把 .ui 里的控件创建到 this 上
        // 可继续连 ui 内控件：connect(buttonBox, &QDialogButtonBox::accepted, ...);
    }
    QString data() const { return lineEdit->text(); }   // 对外暴露取数接口
};
```

约定与标准取数流程：`accept()` 关闭并使 exec() 返回 Accepted（OK 触发）；`reject()` 返回 Rejected（Cancel/Esc）；复杂数据走公开取数函数或信号。

```cpp
MyDialog dlg(this);                       // 模态：栈上 + exec
if (dlg.exec() == QDialog::Accepted)
    QString value = dlg.data();          // 关闭后取数据（对象仍存活）
```

构建系统：qmake 加 `FORMS += dialog.ui`；CMake 把 `.ui` 加进目标源文件并开 `set(CMAKE_AUTOUIC ON)`。

### 9.4 QPen / QBrush / QColor 构造表

### 9.5 自定义控件实战：环形进度条（约 60 行）

```cpp
// ringprogress.h —— paintEvent 里 drawArc 按 value 画弧
#pragma once
#include <QWidget>
#include <QPainter>

class RingProgress : public QWidget {
    Q_OBJECT
public:
    explicit RingProgress(QWidget *parent = nullptr) : QWidget(parent) {
        setMinimumSize(140, 140);            // 给绘制留空间
    }
    int value() const { return m_value; }

public slots:
    void setValue(int v) {                   // 设计成槽：任何信号可直接驱动
        v = qBound(0, v, 100);
        if (v == m_value) return;
        m_value = v;
        emit valueChanged(v);
        update();                            // 投递重绘请求（多次调用自动合并）
    }

signals:
    void valueChanged(int v);

protected:
    void paintEvent(QPaintEvent *) override {
        QPainter p(this);
        p.setRenderHint(QPainter::Antialiasing, true);   // 抗锯齿【高频】

        const int side = qMin(width(), height()) - 20;   // 正方形边长，留 10px 边距
        QRectF circle(-side / 2.0, -side / 2.0, side, side);
        p.translate(width() / 2.0, height() / 2.0);      // 原点移到中心
        p.setBrush(Qt::NoBrush);

        p.setPen(QPen(QColor("#e6e6e6"), 12, Qt::SolidLine, Qt::RoundCap));
        p.drawEllipse(circle);                            // 底环

        // 进度环：12 点方向起，负 span=顺时针；角度单位 1/16 度
        p.setPen(QPen(QColor("#4a90d9"), 12, Qt::SolidLine, Qt::RoundCap));
        p.drawArc(circle, 90 * 16, -int(m_value * 3.6) * 16);

        p.setPen(QColor("#333333"));
        p.setFont(QFont(font().family(), side / 8, QFont::Bold));
        p.drawText(circle, Qt::AlignCenter, QString::number(m_value) + "%");
    }

private:
    int m_value = 0;
};
```

### 9.6 update() vs repaint()【高频】

| | update() | repaint() |
|---|---|---|
| 机制 | 向事件队列**投递**绘制事件，回事件循环统一处理 | **立即同步**调用 paintEvent |
| 合并 | 多次调用自动合并为一个（脏区求并集），高频刷新首选 | 不合并，调几次画几次 |
| 嵌套安全 | paintEvent 内调用安全 | paintEvent 内调用触发 "Recursive repaint detected" 警告 |
| 线程 | 仍属 GUI 操作：控件只属于主线程，跨线程刷新请用信号槽回主线程再调 | 同左，官方文档明确警告不要在非 GUI 线程调用 |

结论：默认永远用 update()；repaint() 仅用于极少数需要"当场画完"的同步场景。

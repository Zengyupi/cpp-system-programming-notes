## 1. 进程资源监控：定时采样CPU/内存/FD并绘图

**场景**：排查C++服务内存泄漏或CPU飙高时，需要长时间记录进程资源变化。脚本定时采样指定PID的CPU、内存、FD数、线程数，输出CSV并用matplotlib绘图，便于事后分析。

```python
#!/usr/bin/env python3
"""进程资源监控：定时采样CPU/内存/FD/线程，输出CSV+图表。"""
import time
import sys
import csv
import argparse
from pathlib import Path
from datetime import datetime

try:
    import psutil  # pip install psutil
except ImportError:
    print("需要 psutil: pip install psutil")
    sys.exit(1)

def sample_process(pid: int, proc=None) -> dict | None:
    """采样一次进程资源。"""
    try:
        if proc is None:
            proc = psutil.Process(pid)
        with proc.oneshot():
            mem = proc.memory_info()
            return {
                "timestamp": datetime.now().strftime("%H:%M:%S.%f")[:-3],
                "cpu_percent": proc.cpu_percent(interval=None),
                "rss_mb": mem.rss / 1024 / 1024,
                "vms_mb": mem.vms / 1024 / 1024,
                "num_fds": proc.num_fds() if hasattr(proc, "num_fds") else 0,
                "num_threads": proc.num_threads(),
                "status": proc.status(),
            }
    except (psutil.NoSuchProcess, psutil.AccessDenied):
        return None

def monitor(pid: int, interval: float, duration: float, output: Path):
    """持续监控并写入CSV。"""
    proc = psutil.Process(pid)
    # 第一次cpu_percent需要调用一次以建立基准
    proc.cpu_percent(interval=None)

    fieldnames = ["timestamp", "cpu_percent", "rss_mb", "vms_mb", "num_fds", "num_threads", "status"]
    end_time = time.time() + duration if duration > 0 else float("inf")
    sample_count = 0

    with open(output, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()

        print(f"监控 PID={pid}, 间隔={interval}s, 输出={output}")
        print("按 Ctrl+C 停止")

        try:
            while time.time() < end_time:
                data = sample_process(pid, proc)
                if data is None:
                    print(f"\n进程 {pid} 已退出")
                    break
                writer.writerow(data)
                f.flush()
                sample_count += 1
                # 实时打印摘要
                print(f"\r  {data['timestamp']}  CPU={data['cpu_percent']:5.1f}%  "
                      f"RSS={data['rss_mb']:7.1f}MB  FD={data['num_fds']:4d}  "
                      f"线程={data['num_threads']:3d}", end="", flush=True)
                time.sleep(interval)
        except KeyboardInterrupt:
            print(f"\n用户中断，已采样 {sample_count} 次")

    print(f"\n已保存 {sample_count} 条采样到 {output}")
    return sample_count

def plot_csv(csv_path: Path, output_png: Path):
    """从CSV生成图表。"""
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
    except ImportError:
        print("matplotlib未安装，跳过绘图。pip install matplotlib")
        return

    timestamps, cpu, rss, fds, threads = [], [], [], [], []
    with open(csv_path, encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            timestamps.append(row["timestamp"])
            cpu.append(float(row["cpu_percent"]))
            rss.append(float(row["rss_mb"]))
            fds.append(int(row["num_fds"]))
            threads.append(int(row["num_threads"]))

    fig, axes = plt.subplots(3, 1, figsize=(12, 8), sharex=True)
    x = range(len(timestamps))

    axes[0].plot(x, cpu, color="red", linewidth=0.8)
    axes[0].set_ylabel("CPU %")
    axes[0].set_title(f"Process Resource Monitor (PID)")
    axes[0].grid(True, alpha=0.3)

    axes[1].plot(x, rss, color="blue", linewidth=0.8)
    axes[1].set_ylabel("RSS MB")
    axes[1].grid(True, alpha=0.3)

    axes[2].plot(x, fds, color="green", linewidth=0.8, label="FD")
    axes[2].plot(x, threads, color="orange", linewidth=0.8, label="Threads")
    axes[2].set_ylabel("Count")
    axes[2].legend()
    axes[2].grid(True, alpha=0.3)

    # x轴刻度
    tick_step = max(1, len(timestamps) // 10)
    axes[2].set_xticks(x[::tick_step])
    axes[2].set_xticklabels(timestamps[::tick_step], rotation=45, fontsize=8)
    axes[2].set_xlabel("Time")

    plt.tight_layout()
    plt.savefig(output_png, dpi=120)
    print(f"图表已保存到 {output_png}")

def main():
    parser = argparse.ArgumentParser(description="进程资源监控")
    parser.add_argument("pid", type=int, help="进程PID")
    parser.add_argument("--interval", type=float, default=1.0, help="采样间隔秒")
    parser.add_argument("--duration", type=float, default=0, help="监控时长秒，0=无限")
    parser.add_argument("--output", default="", help="CSV输出路径")
    parser.add_argument("--plot", action="store_true", help="结束后生成图表")
    args = parser.parse_args()

    output = Path(args.output) if args.output else Path(f"proc_monitor_{args.pid}.csv")
    count = monitor(args.pid, args.interval, args.duration, output)

    if args.plot and count > 0:
        plot_csv(output, output.with_suffix(".png"))

if __name__ == "__main__":
    main()
```

**改造点**：
- `sample_process`：扩展采样指标，如`proc.io_counters()`（磁盘IO）、`proc.net_connections()`（连接数）、`proc.environ()`（环境变量快照）。
- 多进程监控：传入PID列表，对每个进程单独采样，CSV中加`pid`列。
- 告警阈值：RSS超过阈值或FD数持续增长时触发告警（写日志/发邮件/调用webhook）。

**常见坑**：`cpu_percent(interval=None)`第一次调用返回0，需要先调用一次建立基准；`proc.oneshot()`上下文管理器可显著减少系统调用次数，提升采样效率；Linux下`num_fds()`需要`/proc/<pid>/fd`读取权限；长时间监控CSV文件会变大，可加轮转或只保留最近N条。监控发现内存持续增长时，配合《../../01-C++技术体系/07-调试与测试/02-Valgrind内存检测.md》定位泄漏点。

## 2. 发布打包：编译产物+依赖+配置一键打包

**场景**：C++服务发布时需要收集二进制、动态库、配置文件、启动脚本，按目录结构打包成tar.gz并生成版本信息。脚本自动解析二进制依赖（`ldd`/`otool`），复制所需so，生成MD5校验和CHANGELOG。

```python
#!/usr/bin/env python3
"""发布打包：收集二进制+依赖库+配置+脚本，生成tar.gz发布包。"""
import subprocess
import sys
import os
import shutil
import hashlib
import tarfile
import time
from pathlib import Path
from datetime import datetime

# ============ 改造点：打包配置 ============
PROJECT_NAME = "myservice"
VERSION = "1.2.0"
BUILD_DIR = Path("build")
RELEASE_DIR = Path("release")

# 需要打包的文件/目录（源路径 -> 发布包内相对路径）
INCLUDE_FILES = {
    "build/bin/myserver":          "bin/myserver",
    "build/bin/mycli":             "bin/mycli",
    "config/prod.yaml":            "config/prod.yaml",
    "config/log4cxx.properties":   "config/log4cxx.properties",
    "scripts/start.sh":            "scripts/start.sh",
    "scripts/stop.sh":             "scripts/stop.sh",
    "README.md":                   "README.md",
    "CHANGELOG.md":                "CHANGELOG.md",
}

# 额外需要包含的动态库路径（自动依赖之外手动补充）
EXTRA_LIBS = [
    # "/usr/local/lib/libcustom.so.1",
]

# 排除的系统库（不打包这些，目标机应有）
EXCLUDE_LIB_PATTERNS = [
    "linux-vdso", "ld-linux", "libc.so", "libpthread", "libdl",
    "libm.so", "librt", "libresolv", "libutil", "libnsl",
]

def get_dependencies(binary: Path) -> list[Path]:
    """解析Linux二进制的动态库依赖（ldd）。"""
    try:
        result = subprocess.run(
            ["ldd", str(binary)], capture_output=True, text=True, timeout=10
        )
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return []

    deps = []
    for line in result.stdout.splitlines():
        line = line.strip()
        # ldd输出格式: libxxx.so => /path/to/libxxx.so (0x...)
        if "=>" in line:
            parts = line.split("=>")
            lib_path = parts[1].strip().split()[0]
            if lib_path and lib_path != "not" and Path(lib_path).exists():
                deps.append(Path(lib_path))
        elif line.startswith("/"):
            # 直接路径形式
            p = line.split()[0]
            if Path(p).exists():
                deps.append(Path(p))
    return deps

def should_exclude_lib(lib: Path) -> bool:
    name = lib.name
    return any(pat in name for pat in EXCLUDE_LIB_PATTERNS)

def md5_file(path: Path) -> str:
    h = hashlib.md5()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()

def collect_all_dependencies(binaries: list[Path]) -> set[Path]:
    """递归收集所有二进制的依赖（含依赖的依赖）。"""
    all_deps: set[Path] = set()
    to_check = list(binaries)
    checked: set[Path] = set()

    while to_check:
        binary = to_check.pop()
        if binary in checked:
            continue
        checked.add(binary)
        for dep in get_dependencies(binary):
            if should_exclude_lib(dep):
                continue
            if dep not in all_deps:
                all_deps.add(dep)
                to_check.append(dep)  # 递归检查依赖的依赖
    return all_deps

def build_package():
    """构建发布包。"""
    package_name = f"{PROJECT_NAME}-{VERSION}"
    staging = RELEASE_DIR / package_name
    lib_dir = staging / "lib"

    # 清理旧的staging
    if staging.exists():
        shutil.rmtree(staging)
    staging.mkdir(parents=True)
    lib_dir.mkdir()

    # 1. 复制指定文件
    print("## 复制文件")
    binaries = []
    for src_rel, dst_rel in INCLUDE_FILES.items():
        src = Path(src_rel)
        if not src.exists():
            print(f"  跳过(不存在): {src}")
            continue
        dst = staging / dst_rel
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dst)
        print(f"  {src} -> {dst_rel}")
        # 记录二进制文件（无扩展名或在bin目录）
        if dst_rel.startswith("bin/") or (src.is_file() and os.access(src, os.X_OK)):
            binaries.append(dst)

    # 2. 收集并复制依赖库
    print("\n## 收集动态库依赖")
    all_deps = collect_all_dependencies(binaries)
    for extra in EXTRA_LIBS:
        p = Path(extra)
        if p.exists():
            all_deps.add(p)

    for dep in sorted(all_deps):
        dst = lib_dir / dep.name
        shutil.copy2(dep, dst)
        print(f"  {dep.name}")

    # 3. 生成版本信息
    print("\n## 生成版本信息")
    version_info = {
        "project": PROJECT_NAME,
        "version": VERSION,
        "build_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "files": {},
    }
    for f in staging.rglob("*"):
        if f.is_file():
            rel = f.relative_to(staging)
            version_info["files"][str(rel)] = {
                "md5": md5_file(f),
                "size": f.stat().st_size,
            }
    (staging / "VERSION.json").write_text(
        __import__("json").dumps(version_info, indent=2, ensure_ascii=False),
        encoding="utf-8",
    )

    # 4. 打包tar.gz
    print("\n## 打包")
    tar_path = RELEASE_DIR / f"{package_name}.tar.gz"
    with tarfile.open(tar_path, "w:gz") as tar:
        tar.add(staging, arcname=package_name)

    tar_md5 = md5_file(tar_path)
    print(f"  发布包: {tar_path}")
    print(f"  大小: {tar_path.stat().st_size / 1024 / 1024:.2f} MB")
    print(f"  MD5: {tar_md5}")

    # 5. 生成MD5校验文件
    (tar_path.with_name(tar_path.name + ".md5")).write_text(f"{tar_md5}  {tar_path.name}\n", encoding="utf-8")

    # 清理staging
    shutil.rmtree(staging)
    print(f"\n完成！发布包位于 {tar_path}")

def main():
    if not BUILD_DIR.exists():
        print(f"构建目录不存在: {BUILD_DIR}，请先编译")
        sys.exit(1)
    RELEASE_DIR.mkdir(exist_ok=True)
    build_package()

if __name__ == "__main__":
    main()
```

**改造点**：
- `INCLUDE_FILES`：按项目实际结构调整源路径和发布包内路径映射。
- `get_dependencies`：macOS下替换为`otool -L`解析，Windows下用`dumpbin /dependents`或`Dependencies`工具。
- `EXCLUDE_LIB_PATTERNS`：根据目标机环境调整，确保不打包目标机已有的系统库。
- 打包格式：当前`tar.gz`，可改为`zip`（`shutil.make_archive`）或`deb`/`rpm`（需`dpkg-deb`/`rpmbuild`）。

**常见坑**：`ldd`输出中`linux-vdso.so`是虚拟动态库，没有实际文件，必须排除；复制动态库时用`shutil.copy2`保留权限和时间戳；软链接库（如`libfoo.so -> libfoo.so.1.2`）需要同时复制链接和目标文件，或直接复制解析后的真实路径；打包后验证`LD_LIBRARY_PATH=./lib ./bin/myserver`能正常启动。

## 3. Git批量操作：多仓库批量拉取/切分支/查状态

**场景**：大型项目由多个Git仓库组成（monorepo拆分或微服务架构），需要批量拉取最新代码、切换统一分支、检查未提交变更、查看各仓库当前分支。脚本并行操作多仓库，输出状态矩阵，异常仓库单独标记。

```python
#!/usr/bin/env python3
"""Git批量操作：多仓库并行拉取/切分支/查状态/清理。"""
import subprocess
import sys
import argparse
import concurrent.futures
from pathlib import Path
from dataclasses import dataclass

@dataclass
class RepoStatus:
    path: str
    branch: str = ""
    ahead: int = 0
    behind: int = 0
    dirty: bool = False
    untracked: int = 0
    last_commit: str = ""
    error: str = ""

def run_git(repo: Path, args: list[str], timeout: int = 30) -> tuple[int, str, str]:
    """在指定仓库执行git命令。"""
    cmd = ["git", "-C", str(repo)] + args
    try:
        proc = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        return proc.returncode, proc.stdout.strip(), proc.stderr.strip()
    except subprocess.TimeoutExpired:
        return -1, "", "TIMEOUT"

def is_git_repo(path: Path) -> bool:
    return (path / ".git").exists()

def find_repos(root: Path, max_depth: int = 3) -> list[Path]:
    """递归查找Git仓库。"""
    repos = []
    for p in root.rglob(".git"):
        repo = p.parent
        # 检查深度
        try:
            rel = repo.relative_to(root)
            if len(rel.parts) <= max_depth:
                repos.append(repo)
        except ValueError:
            pass
    return sorted(repos)

def get_status(repo: Path) -> RepoStatus:
    """获取仓库状态。"""
    status = RepoStatus(path=str(repo))

    # 当前分支
    rc, out, err = run_git(repo, ["rev-parse", "--abbrev-ref", "HEAD"])
    if rc != 0:
        status.error = err or "无法获取分支"
        return status
    status.branch = out

    # 远程跟踪分支的ahead/behind
    rc, out, _ = run_git(repo, ["rev-list", "--left-right", "--count", "@{u}...HEAD"])
    if rc == 0 and out:
        parts = out.split()
        if len(parts) == 2:
            status.behind = int(parts[0])
            status.ahead = int(parts[1])

    # 是否有未提交变更
    rc, out, _ = run_git(repo, ["status", "--porcelain"])
    if rc == 0:
        lines = out.splitlines() if out else []
        status.dirty = any(not line.startswith("??") for line in lines)
        status.untracked = sum(1 for line in lines if line.startswith("??"))

    # 最后一次提交
    rc, out, _ = run_git(repo, ["log", "-1", "--format=%h %s (%cr)"])
    if rc == 0:
        status.last_commit = out[:80]

    return status

def pull_repo(repo: Path, rebase: bool = False) -> str:
    """拉取仓库。返回结果描述。"""
    args = ["pull", "--rebase"] if rebase else ["pull"]
    rc, out, err = run_git(repo, args, timeout=120)
    if rc == 0:
        return "OK" if "Already up to date" in out else "已更新"
    if "CONFLICT" in err or "CONFLICT" in out:
        return "冲突!"
    if rc == -1:
        return "超时"
    return f"失败: {err[:60]}"

def checkout_repo(repo: Path, branch: str, create: bool = False) -> str:
    """切换分支。"""
    if create:
        rc, out, err = run_git(repo, ["checkout", "-b", branch])
    else:
        rc, out, err = run_git(repo, ["checkout", branch])
    if rc == 0:
        return f"已切到 {branch}"
    return f"失败: {err[:80]}"

def fetch_repo(repo: Path) -> str:
    """fetch所有远程。"""
    rc, out, err = run_git(repo, ["fetch", "--all", "--prune"], timeout=60)
    return "OK" if rc == 0 else f"失败: {err[:60]}"

def main():
    parser = argparse.ArgumentParser(description="Git批量操作")
    parser.add_argument("root", help="仓库根目录（包含多个git仓库）")
    parser.add_argument("--action", choices=["status", "pull", "fetch", "checkout"],
                        default="status", help="操作类型")
    parser.add_argument("--branch", default="", help="checkout目标分支")
    parser.add_argument("--create", action="store_true", help="创建新分支")
    parser.add_argument("--rebase", action="store_true", help="pull时使用rebase")
    parser.add_argument("--jobs", type=int, default=4, help="并行数")
    parser.add_argument("--depth", type=int, default=3, help="查找仓库最大深度")
    args = parser.parse_args()

    root = Path(args.root)
    if not root.is_dir():
        print(f"目录不存在: {root}")
        sys.exit(1)

    repos = find_repos(root, args.depth)
    if not repos:
        print("未找到Git仓库")
        sys.exit(1)
    print(f"找到 {len(repos)} 个仓库，操作: {args.action}\n")

    if args.action == "status":
        # 并行获取状态
        with concurrent.futures.ThreadPoolExecutor(max_workers=args.jobs) as executor:
            statuses = list(executor.map(get_status, repos))

        # 打印状态矩阵
        print(f"{'仓库':<30} {'分支':<20} {'领先':<5} {'落后':<5} {'变更':<5} {'未跟踪':<6}")
        print("-" * 85)
        dirty_count = 0
        for s in statuses:
            if s.error:
                print(f"{s.path[-28:]:<30} 错误: {s.error}")
                continue
            dirty_flag = "✗" if s.dirty else "✓"
            if s.dirty:
                dirty_count += 1
            print(f"{Path(s.path).name[-28:]:<30} {s.branch[-18:]:<20} "
                  f"{s.ahead:<5} {s.behind:<5} {dirty_flag:<5} {s.untracked:<6}")
        print(f"\n有未提交变更的仓库: {dirty_count}/{len(repos)}")

    elif args.action == "pull":
        with concurrent.futures.ThreadPoolExecutor(max_workers=args.jobs) as executor:
            results = list(executor.map(lambda r: pull_repo(r, args.rebase), repos))
        for repo, result in zip(repos, results):
            print(f"  {repo.name:<30} {result}")

    elif args.action == "fetch":
        with concurrent.futures.ThreadPoolExecutor(max_workers=args.jobs) as executor:
            results = list(executor.map(fetch_repo, repos))
        for repo, result in zip(repos, results):
            print(f"  {repo.name:<30} {result}")

    elif args.action == "checkout":
        if not args.branch:
            print("checkout操作需要 --branch 指定分支")
            sys.exit(1)
        with concurrent.futures.ThreadPoolExecutor(max_workers=args.jobs) as executor:
            results = list(executor.map(
                lambda r: checkout_repo(r, args.branch, args.create), repos
            ))
        for repo, result in zip(repos, results):
            print(f"  {repo.name:<30} {result}")

if __name__ == "__main__":
    main()
```

**改造点**：
- `find_repos`：如果仓库列表是固定的，可直接从配置文件读取路径列表，而非递归扫描。
- `get_status`：扩展状态信息，如`git stash list`检查暂存、`git log`检查最近提交者。
- 批量提交：增加`commit`操作，对所有有变更的仓库执行统一的`git add -A && git commit -m "msg"`。
- 批量推送：增加`push`操作，推送所有ahead的仓库。

**常见坑**：`git pull`在有未提交变更时会失败，批量pull前先检查dirty状态；`@{u}`（上游跟踪分支）在本地分支未设置上游时报错，需要先`git branch -u origin/branch`；并行操作同一文件系统的多个仓库可能IO瓶颈，`--jobs`不宜过大；`git status --porcelain`输出中`??`是未跟踪文件，`M`/`A`/`D`等是已跟踪文件变更，区分处理。

## 4. 定时巡检：服务存活+端口+日志错误巡检告警

**场景**：线上部署多个C++服务，需要定时巡检服务进程是否存活、端口是否监听、日志是否出现ERROR、磁盘/内存是否超限。脚本按配置巡检所有服务，异常时输出告警报告（可对接邮件/钉钉/企业微信webhook）。

```python
#!/usr/bin/env python3
"""定时巡检：服务存活+端口+日志错误+系统资源，异常告警。"""
import socket
import subprocess
import re
import time
import sys
import json
import argparse
from pathlib import Path
from dataclasses import dataclass, field
from datetime import datetime

@dataclass
class ServiceCheck:
    name: str
    process_pattern: str = ""      # 进程匹配（pgrep模式）
    port: int = 0                   # 监听端口
    log_file: str = ""              # 日志文件路径
    error_pattern: str = r"ERROR|FATAL|CRITICAL"  # 日志错误正则
    check_interval: int = 60        # 巡检间隔秒

@dataclass
class CheckResult:
    name: str
    process_ok: bool = True
    port_ok: bool = True
    log_errors: int = 0
    error_samples: list[str] = field(default_factory=list)
    message: str = ""

def check_process(pattern: str) -> tuple[bool, str]:
    """检查进程是否存活。"""
    try:
        result = subprocess.run(
            ["pgrep", "-f", pattern], capture_output=True, text=True, timeout=5
        )
        pids = result.stdout.strip().splitlines()
        if pids:
            return True, f"PID={','.join(pids[:5])}"
        return False, "进程不存在"
    except FileNotFoundError:
        # pgrep不可用，用ps替代
        try:
            result = subprocess.run(
                ["ps", "aux"], capture_output=True, text=True, timeout=5
            )
            for line in result.stdout.splitlines():
                if pattern in line and "grep" not in line:
                    return True, "进程存在"
            return False, "进程不存在"
        except Exception as e:
            return False, f"检查失败: {e}"

def check_port(host: str, port: int, timeout: float = 3.0) -> tuple[bool, str]:
    """检查端口是否可连接。"""
    if port == 0:
        return True, "跳过"
    try:
        with socket.create_connection((host, port), timeout=timeout):
            return True, "端口可连接"
    except (ConnectionRefusedError, socket.timeout, OSError) as e:
        return False, f"端口不可达: {e}"

def check_log_errors(log_file: str, pattern: str, since_seconds: int = 300,
                     max_samples: int = 5) -> tuple[int, list[str]]:
    """检查最近N秒日志中的错误。"""
    if not log_file:
        return 0, []
    path = Path(log_file)
    if not path.exists():
        return -1, [f"日志文件不存在: {log_file}"]

    error_re = re.compile(pattern)
    errors = 0
    samples = []
    cutoff = time.time() - since_seconds

    try:
        # 从文件末尾向前读，只处理最近的内容
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            # 简单策略：读取最后1000行（大日志优化用seek+二分）
            lines = f.readlines()[-1000:]
            for line in lines:
                if error_re.search(line):
                    errors += 1
                    if len(samples) < max_samples:
                        samples.append(line.strip()[:200])
    except OSError as e:
        return -1, [f"读取失败: {e}"]

    return errors, samples

def check_disk_usage(path: str = "/", threshold: float = 90.0) -> tuple[bool, str]:
    """检查磁盘使用率。"""
    try:
        result = subprocess.run(["df", "-h", path], capture_output=True, text=True, timeout=5)
        lines = result.stdout.strip().splitlines()
        if len(lines) >= 2:
            parts = lines[1].split()
            usage_str = parts[-2] if len(parts) >= 2 else "0%"
            usage = float(usage_str.rstrip("%"))
            if usage >= threshold:
                return False, f"磁盘使用率 {usage}% >= {threshold}%"
            return True, f"磁盘使用率 {usage}%"
    except Exception as e:
        return False, f"检查失败: {e}"
    return True, "未知"

def check_memory_usage(threshold: float = 90.0) -> tuple[bool, str]:
    """检查内存使用率。"""
    try:
        result = subprocess.run(["free", "-m"], capture_output=True, text=True, timeout=5)
        lines = result.stdout.strip().splitlines()
        for line in lines:
            if line.startswith("Mem:"):
                parts = line.split()
                total = int(parts[1])
                available = int(parts[6]) if len(parts) > 6 else int(parts[3])
                usage = (1 - available / total) * 100
                if usage >= threshold:
                    return False, f"内存使用率 {usage:.1f}% >= {threshold}%"
                return True, f"内存使用率 {usage:.1f}%"
    except Exception as e:
        return False, f"检查失败: {e}"
    return True, "未知"

def run_checks(services: list[ServiceCheck], host: str = "127.0.0.1") -> list[CheckResult]:
    """执行所有巡检。"""
    results = []
    for svc in services:
        result = CheckResult(name=svc.name)

        # 进程检查
        if svc.process_pattern:
            ok, msg = check_process(svc.process_pattern)
            result.process_ok = ok
            if not ok:
                result.message += f"进程异常: {msg}; "

        # 端口检查
        if svc.port:
            ok, msg = check_port(host, svc.port)
            result.port_ok = ok
            if not ok:
                result.message += f"端口异常: {msg}; "

        # 日志错误检查
        if svc.log_file:
            errors, samples = check_log_errors(svc.log_file, svc.error_pattern)
            result.log_errors = errors
            result.error_samples = samples
            if errors > 0:
                result.message += f"日志错误 {errors} 条; "

        results.append(result)
    return results

def print_report(results: list[CheckResult], disk_ok: bool, disk_msg: str,
                 mem_ok: bool, mem_msg: str):
    """打印巡检报告。"""
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    print(f"{'='*60}")
    print(f"巡检报告  {now}")
    print(f"{'='*60}")

    all_ok = True
    for r in results:
        ok = r.process_ok and r.port_ok and r.log_errors <= 0
        status = "✓ 正常" if ok else "✗ 异常"
        if not ok:
            all_ok = False
        print(f"\n[{status}] {r.name}")
        if not r.process_ok:
            print(f"  进程: ✗")
        if not r.port_ok:
            print(f"  端口: ✗")
        if r.log_errors > 0:
            print(f"  日志错误: {r.log_errors} 条")
            for s in r.error_samples[:3]:
                print(f"    > {s}")
        if r.message:
            print(f"  详情: {r.message}")

    print(f"\n系统资源:")
    print(f"  磁盘: {'✓' if disk_ok else '✗'} {disk_msg}")
    print(f"  内存: {'✓' if mem_ok else '✗'} {mem_msg}")

    print(f"\n{'='*60}")
    print("全部正常" if all_ok and disk_ok and mem_ok else "存在异常，请处理！")
    print(f"{'='*60}")

def load_config(config_path: Path) -> tuple[list[ServiceCheck], dict]:
    """加载巡检配置。"""
    config = json.loads(config_path.read_text(encoding="utf-8"))
    services = []
    for s in config.get("services", []):
        services.append(ServiceCheck(
            name=s["name"],
            process_pattern=s.get("process_pattern", ""),
            port=s.get("port", 0),
            log_file=s.get("log_file", ""),
            error_pattern=s.get("error_pattern", r"ERROR|FATAL|CRITICAL"),
        ))
    system = config.get("system", {})
    return services, system

def main():
    parser = argparse.ArgumentParser(description="服务定时巡检")
    parser.add_argument("config", help="巡检配置JSON")
    parser.add_argument("--watch", action="store_true", help="持续巡检模式")
    parser.add_argument("--interval", type=int, default=60, help="巡检间隔秒")
    args = parser.parse_args()

    config_path = Path(args.config)
    if not config_path.exists():
        print(f"配置不存在: {config_path}")
        print("\n配置示例:")
        example = {
            "services": [
                {"name": "网关服务", "process_pattern": "gateway_server", "port": 8080,
                 "log_file": "/var/log/gateway.log"},
                {"name": "数据服务", "process_pattern": "data_server", "port": 9000,
                 "log_file": "/var/log/data.log", "error_pattern": "ERROR|panic"},
            ],
            "system": {"disk_threshold": 90, "mem_threshold": 90},
        }
        print(json.dumps(example, indent=2, ensure_ascii=False))
        sys.exit(1)

    services, system_cfg = load_config(config_path)
    disk_threshold = system_cfg.get("disk_threshold", 90)
    mem_threshold = system_cfg.get("mem_threshold", 90)

    def do_check():
        results = run_checks(services)
        disk_ok, disk_msg = check_disk_usage(threshold=disk_threshold)
        mem_ok, mem_msg = check_memory_usage(threshold=mem_threshold)
        print_report(results, disk_ok, disk_msg, mem_ok, mem_msg)
        # 返回是否有异常
        return all(r.process_ok and r.port_ok and r.log_errors <= 0 for r in results) and disk_ok and mem_ok

    if args.watch:
        print(f"持续巡检模式，间隔 {args.interval}s，Ctrl+C 停止\n")
        try:
            while True:
                all_ok = do_check()
                if not all_ok:
                    # 此处可对接告警：邮件/钉钉/企业微信webhook
                    print("[告警] 检测到异常！")
                time.sleep(args.interval)
                print("\n")
        except KeyboardInterrupt:
            print("\n巡检停止")
    else:
        all_ok = do_check()
        sys.exit(0 if all_ok else 1)

if __name__ == "__main__":
    main()
```

**改造点**：
- `run_checks`：扩展检查项，如HTTP健康检查（`curl /health`）、数据库连接检查、消息队列堆积检查。
- 告警通道：异常时调用webhook（钉钉/企业微信/飞书），或发送邮件（`smtplib`），或写入告警文件供监控系统采集。
- `check_log_errors`：大日志文件优化——用`seek`从文件末尾向前二分查找时间戳，而非读取最后1000行。
- 巡检结果持久化：每次结果写入SQLite或CSV，便于事后分析异常频率和趋势。

**常见坑**：`pgrep -f`匹配的是完整命令行，可能匹配到grep自身或其他包含该字符串的进程，用更精确的模式（如完整路径）；端口检查用`socket.create_connection`会真正建立TCP连接，某些服务可能记录连接日志，可改为只检查`/proc/net/tcp`中的监听状态；`free`命令输出格式在不同Linux发行版有差异，`/proc/meminfo`更可靠；持续巡检模式下日志文件可能被logrotate截断，需要处理文件inode变化的情况。

## 6. 常见坑

- **外部命令不检查返回码**：`subprocess.run` 不加 `check=True` 时失败会被当成功；所有外呼都要查返回码。
- **巡检项互相掩盖失败**：一项异常导致后续全部跳过；每项独立 `try/except` 并汇总，最后统一非 0 退出。
- **logrotate 截断后 inode 变化**：长期持有旧 fd 的进程读不到新内容；用 `copytruncate` 或重新打开文件。
- **打包漏依赖 / 未写版本号**：包内必须写入版本号与 commit，否则线上出问题无法定位版本。
- **阈值用相对值**：`"比上次高"` 这类相对阈值在抖动下会误报；用绝对值 + 连续 N 次确认。
- **cron/systemd timer 环境不同**：定时任务的 PATH 与登录 shell 不同，脚本里用绝对路径并显式设置环境。
- **`psutil.cpu_percent()` 首次返回 0**：先调用一次预热再采样，否则数据无意义。

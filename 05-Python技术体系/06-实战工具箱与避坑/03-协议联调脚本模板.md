## 1. 报文构造器：按协议字段拼装二进制报文

**场景**：调试C++网络服务时需要手动构造特定协议报文（Modbus、IEC104、自定义TCP协议），用hex字符串拼容易出错。脚本用`struct`按字段定义拼装，支持大小端、校验和、CRC，输出hex和原始字节。

```python
#!/usr/bin/env python3
"""报文构造器：按协议字段定义拼装二进制报文，支持校验和/CRC。"""
import struct
import sys
import binascii
from dataclasses import dataclass, field
from typing import Any

# ============ 改造点：协议字段定义 ============
# 示例：一个自定义TCP协议头 + payload
# 字段格式: (名称, struct格式符, 值)
# struct格式: >大端 <小端 !网络序; B字节 H两字节 I四字节 Q八字节 s字符串
@dataclass
class Field:
    name: str
    fmt: str       # struct格式符，如 ">I", "<H", "B", "16s"
    value: Any     # 整数值或bytes

PROTOCOL_FIELDS = [
    Field("magic",      ">H",  0xAA55),      # 魔数
    Field("version",    "B",   1),            # 版本
    Field("msg_type",   "B",   0x03),         # 消息类型
    Field("seq",        ">I",  1001),         # 序列号
    Field("payload_len",">H",  0),            # payload长度（自动计算）
    # payload部分
    Field("cmd",        "B",   0x10),
    Field("data",       "4s",  b"\x01\x02\x03\x04"),
]

def crc16(data: bytes, poly: int = 0xA001, init: int = 0xFFFF) -> int:
    """CRC-16/MODBUS校验。"""
    crc = init
    for byte in data:
        crc ^= byte
        for _ in range(8):
            if crc & 1:
                crc = (crc >> 1) ^ poly
            else:
                crc >>= 1
    return crc & 0xFFFF

def checksum8(data: bytes) -> int:
    """简单8位累加和校验。"""
    return sum(data) & 0xFF

def build_packet(fields: list[Field], auto_len_field: str = "payload_len",
                 crc_field: str | None = None) -> bytes:
    """
    按字段列表拼装报文。
    auto_len_field: 哪个字段的值需要自动设为后续payload的总长度
    crc_field: 哪个字段是CRC（追加在末尾，值自动计算）
    """
    # 先计算payload长度（auto_len_field之后的所有字段）
    payload_bytes = b""
    len_idx = None
    for i, f in enumerate(fields):
        if f.name == auto_len_field:
            len_idx = i
            break
    if len_idx is not None:
        for f in fields[len_idx + 1:]:
            payload_bytes += _pack_field(f)
        fields[len_idx].value = len(payload_bytes)

    # 拼装全部字段
    result = b""
    for f in fields:
        result += _pack_field(f)

    # 追加CRC
    if crc_field:
        crc_val = crc16(result)
        result += struct.pack(">H", crc_val)

    return result

def _pack_field(f: Field) -> bytes:
    if isinstance(f.value, bytes):
        return struct.pack(f.fmt, f.value)
    return struct.pack(f.fmt, f.value)

def hexdump(data: bytes, width: int = 16) -> str:
    """输出格式化hex dump。"""
    lines = []
    for i in range(0, len(data), width):
        chunk = data[i:i+width]
        hex_part = " ".join(f"{b:02X}" for b in chunk)
        ascii_part = "".join(chr(b) if 32 <= b < 127 else "." for b in chunk)
        lines.append(f"  {i:04X}: {hex_part:<{width*3}}  {ascii_part}")
    return "\n".join(lines)

def main():
    # 构造报文
    packet = build_packet(PROTOCOL_FIELDS, auto_len_field="payload_len", crc_field="crc")

    print("## 字段值")
    for f in PROTOCOL_FIELDS:
        val = f.value.hex() if isinstance(f.value, bytes) else f"0x{f.value:X}"
        print(f"  {f.name:<15} = {val}")

    print(f"\n## 报文总长度: {len(packet)} 字节")
    print(f"## Hex字符串: {packet.hex().upper()}")
    print(f"## Hex dump:")
    print(hexdump(packet))

    # 保存到文件供发送
    out = sys.argv[1] if len(sys.argv) > 1 else "packet.bin"
    with open(out, "wb") as f:
        f.write(packet)
    print(f"\n已保存到 {out}")
    print(f"发送命令示例: cat {out} | nc -w1 127.0.0.1 8080")

if __name__ == "__main__":
    main()
```

**改造点**：
- `PROTOCOL_FIELDS`：按实际协议定义字段列表，`struct`格式符`>`大端/`<`小端/`!`网络序，`B/H/I/Q`对应1/2/4/8字节无符号整数，`Ns`为N字节字符串。
- `crc16`函数：不同协议CRC多项式不同（Modbus用0xA001，CCITT用0x1021），按需修改。
- `auto_len_field`：自动计算长度字段，适用于header中带payload length的协议。

**常见坑**：`struct.pack`的格式符中`s`类型需要传`bytes`而非`str`；大小端混淆是最常见错误，网络协议默认大端（`!`或`>`）；CRC计算范围要确认是否包含长度字段本身；多字节字段对齐问题，`struct`默认不填充（native模式除外），用标准模式（`>`/`<`/`!`）避免对齐意外。构造的报文可配合《../../02-网络协议与报文解析/08-抓包分析与手写报文解析.md》中的tcpdump/Wireshark验证字节序与校验和。

## 2. mock服务端：模拟对端协议行为用于联调

**场景**：C++客户端开发时对端服务尚未就绪，或需要模拟特定异常场景（超时、错误码、慢响应）。脚本实现一个可配置的TCP/UDP mock服务端，按规则匹配请求并返回预设响应，支持延迟、随机丢包、序列响应。

```python
#!/usr/bin/env python3
"""Mock服务端：TCP/UDP模拟对端，按规则匹配请求返回预设响应。"""
import socket
import struct
import time
import json
import threading
import sys
import random
from pathlib import Path
from dataclasses import dataclass, field

@dataclass
class MockRule:
    name: str
    match_type: str = "contains"   # contains / startswith / regex / hex
    match_pattern: str = ""
    responses: list[dict] = field(default_factory=list)  # [{data, delay_ms, probability}]
    call_count: int = 0

def load_rules(config_path: Path) -> list[MockRule]:
    """从JSON加载mock规则。"""
    config = json.loads(config_path.read_text(encoding="utf-8"))
    rules = []
    for r in config.get("rules", []):
        rule = MockRule(
            name=r["name"],
            match_type=r.get("match_type", "contains"),
            match_pattern=r.get("match_pattern", ""),
            responses=r.get("responses", []),
        )
        rules.append(rule)
    return rules

def match_request(rule: MockRule, data: bytes) -> bool:
    """检查请求是否匹配规则。"""
    if rule.match_type == "contains":
        return rule.match_pattern.encode() in data
    elif rule.match_type == "startswith":
        return data.startswith(rule.match_pattern.encode())
    elif rule.match_type == "hex":
        # match_pattern是hex字符串，如 "AA5503"
        pattern_bytes = bytes.fromhex(rule.match_pattern)
        return data.startswith(pattern_bytes) or pattern_bytes in data
    elif rule.match_type == "regex":
        import re
        return bool(re.search(rule.match_pattern, data.decode("utf-8", errors="replace")))
    return False

def select_response(rule: MockRule) -> tuple[bytes, int]:
    """根据概率选择响应，支持序列模式（按调用顺序依次返回）。"""
    rule.call_count += 1
    if not rule.responses:
        return b"", 0

    # 序列模式：第一个响应带"sequence": true
    if rule.responses[0].get("sequence"):
        idx = (rule.call_count - 1) % len(rule.responses)
        resp = rule.responses[idx]
    else:
        # 概率模式
        r = random.random()
        cumulative = 0.0
        selected = rule.responses[-1]  # 默认最后一个
        for resp in rule.responses:
            cumulative += resp.get("probability", 1.0)
            if r <= cumulative:
                selected = resp
                break

    # 响应数据：hex字符串或普通字符串
    data_str = selected.get("data", "")
    if selected.get("hex", False):
        resp_data = bytes.fromhex(data_str)
    else:
        resp_data = data_str.encode()
    delay = selected.get("delay_ms", 0)
    return resp_data, delay

def handle_client(conn: socket.socket, addr, rules: list[MockRule], default_response: bytes):
    """处理单个TCP客户端连接。"""
    print(f"[连接] {addr}")
    try:
        while True:
            data = conn.recv(4096)
            if not data:
                break
            print(f"[收到] {addr} {len(data)}字节: {data[:64].hex()}")

            # 匹配规则
            resp_data, delay = default_response, 0
            for rule in rules:
                if match_request(rule, data):
                    resp_data, delay = select_response(rule)
                    print(f"  匹配规则: {rule.name} (第{rule.call_count}次)")
                    break

            if delay > 0:
                time.sleep(delay / 1000.0)
            if resp_data:
                conn.sendall(resp_data)
                print(f"  发送: {len(resp_data)}字节")
    except (ConnectionResetError, BrokenPipeError):
        pass
    finally:
        conn.close()
        print(f"[断开] {addr}")

def run_tcp_server(host: str, port: int, rules: list[MockRule], default_response: bytes):
    server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server.bind((host, port))
    server.listen(16)
    print(f"Mock TCP服务端监听 {host}:{port}")

    try:
        while True:
            conn, addr = server.accept()
            t = threading.Thread(target=handle_client, args=(conn, addr, rules, default_response), daemon=True)
            t.start()
    except KeyboardInterrupt:
        print("\n服务端停止")
    finally:
        server.close()

def run_udp_server(host: str, port: int, rules: list[MockRule], default_response: bytes):
    server = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    server.bind((host, port))
    print(f"Mock UDP服务端监听 {host}:{port}")
    try:
        while True:
            data, addr = server.recvfrom(4096)
            print(f"[UDP收到] {addr} {len(data)}字节")
            resp_data, delay = default_response, 0
            for rule in rules:
                if match_request(rule, data):
                    resp_data, delay = select_response(rule)
                    break
            if delay > 0:
                time.sleep(delay / 1000.0)
            if resp_data:
                server.sendto(resp_data, addr)
    except KeyboardInterrupt:
        print("\n服务端停止")
    finally:
        server.close()

def main():
    if len(sys.argv) < 2:
        print("用法: python3 mock_server.py <config.json> [--udp]")
        print("\n配置示例:")
        example = {
            "host": "0.0.0.0",
            "port": 9000,
            "default_response": {"data": "error: no rule matched", "hex": False},
            "rules": [
                {
                    "name": "heartbeat",
                    "match_type": "hex",
                    "match_pattern": "AA5501",
                    "responses": [{"data": "AA550100", "hex": True, "delay_ms": 5, "probability": 1.0}]
                },
                {
                    "name": "flaky_api",
                    "match_type": "contains",
                    "match_pattern": "GET /data",
                    "responses": [
                        {"data": '{"code":0,"data":"ok"}', "probability": 0.7},
                        {"data": '{"code":500,"msg":"internal error"}', "probability": 0.2, "delay_ms": 2000},
                        {"data": "", "probability": 0.1},
                    ]
                }
            ]
        }
        print(json.dumps(example, indent=2, ensure_ascii=False))
        sys.exit(1)

    config_path = Path(sys.argv[1])
    config = json.loads(config_path.read_text(encoding="utf-8"))
    rules = load_rules(config_path)

    default = config.get("default_response", {})
    default_response = bytes.fromhex(default["data"]) if default.get("hex") else default.get("data", "").encode()

    host = config.get("host", "0.0.0.0")
    port = config.get("port", 9000)

    if "--udp" in sys.argv:
        run_udp_server(host, port, rules, default_response)
    else:
        run_tcp_server(host, port, rules, default_response)

if __name__ == "__main__":
    main()
```

**改造点**：
- `match_request`：扩展匹配方式，如按报文偏移量匹配特定字段值（`struct.unpack`解析后比对）。
- `select_response`：增加动态响应逻辑（从请求中提取字段后构造响应），而非仅静态预设。
- 协议支持：当前是裸TCP/UDP，可扩展为HTTP mock（用`http.server`）、Modbus TCP mock等。

**常见坑**：TCP粘包问题——`recv(4096)`可能一次收到多个请求或半个请求，简单mock可忽略，严格场景需要按协议边界拆包；`sendall` vs `send`——必须用`sendall`确保全部发送；多线程下`rule.call_count`不是原子的，精确计数需要加锁；UDP无连接，`recvfrom`一次只能收一个数据报。

## 4. 常见坑

- **字节序 / 长度字段搞反**：先用抓包样本逐字节对齐再写逻辑，别凭协议文档"想当然"。
- **一次改动多个字段**：出问题无法定位是哪个字段错了；一次一个字段，小步验证。
- **mock 服务端没处理粘包/半包**：TCP 是字节流，必须按长度字段循环 `recv_exact(n)`，不能假设一包一收。
- **只发不收、不校验响应**：错误被静默吞掉；构造→发送→解析→断言四步都要有。
- **端口被占用**：`bind` 失败要显式报错退出，而不是继续跑一个"假联调"。
- **序号 / 校验和回填错误**：这类字段最容易写错且对端会直接丢弃，务必单独断言。
- **超时设得过短或过长**：过短导致误判失败，过长掩盖真实问题；按对端响应分布设合理值。

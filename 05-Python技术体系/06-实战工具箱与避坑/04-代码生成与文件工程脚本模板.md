## 1. 批量重命名：按规则统一文件名与序号

**场景**：抓包文件、core文件、测试截图等命名混乱，需要按统一规则重命名（加前缀、补零序号、替换关键字、按时间排序）。脚本支持预览模式（dry-run），确认无误后再执行。

```python
#!/usr/bin/env python3
"""批量重命名：支持前缀/序号/替换/按时间排序，dry-run预览。"""
import re
import sys
import argparse
from pathlib import Path
from datetime import datetime

def build_renames(directory: Path, pattern: str, args) -> list[tuple[Path, Path]]:
    """
    生成重命名映射列表。
    pattern: 匹配文件的glob模式，如 "*.pcap", "core.*"
    """
    files = sorted(directory.glob(pattern), key=lambda p: p.stat().st_mtime)
    renames = []
    for idx, src in enumerate(files, start=args.start):
        # 构造新文件名
        name_parts = []
        if args.prefix:
            name_parts.append(args.prefix)
        if args.number:
            name_parts.append(f"{idx:0{args.zfill}d}")
        # 原始文件名（去扩展名）或替换后的
        stem = src.stem
        if args.replace_from and args.replace_to is not None:
            stem = stem.replace(args.replace_from, args.replace_to)
        if args.regex_search and args.regex_replace is not None:
            stem = re.sub(args.regex_search, args.regex_replace, stem)
        if stem and not args.discard_original:
            name_parts.append(stem)

        new_stem = "_".join(name_parts) if name_parts else src.stem
        dst = src.with_name(new_stem + src.suffix)
        if src != dst:
            renames.append((src, dst))
    return renames

def main():
    parser = argparse.ArgumentParser(description="批量重命名工具")
    parser.add_argument("directory", help="目标目录")
    parser.add_argument("pattern", help="glob匹配模式，如 *.pcap")
    parser.add_argument("--prefix", default="", help="文件名前缀")
    parser.add_argument("--number", action="store_true", help="添加序号")
    parser.add_argument("--start", type=int, default=1, help="序号起始值")
    parser.add_argument("--zfill", type=int, default=3, help="序号补零位数")
    parser.add_argument("--replace-from", default="", help="替换原始名中的字符串")
    parser.add_argument("--replace-to", default="", help="替换为")
    parser.add_argument("--regex-search", default="", help="正则搜索")
    parser.add_argument("--regex-replace", default="", help="正则替换")
    parser.add_argument("--discard-original", action="store_true", help="丢弃原始文件名")
    parser.add_argument("--execute", action="store_true", help="实际执行（默认dry-run）")
    args = parser.parse_args()

    directory = Path(args.directory)
    if not directory.is_dir():
        print(f"目录不存在: {directory}")
        sys.exit(1)

    renames = build_renames(directory, args.pattern, args)
    if not renames:
        print("没有需要重命名的文件")
        return

    print(f"共 {len(renames)} 个文件将重命名:")
    for src, dst in renames:
        print(f"  {src.name}  ->  {dst.name}")

    # 检查目标冲突
    dst_names = [d.name for _, d in renames]
    conflicts = set(n for n in dst_names if dst_names.count(n) > 1)
    if conflicts:
        print(f"\n警告: 目标文件名冲突: {conflicts}")
        sys.exit(1)

    if args.execute:
        for src, dst in renames:
            src.rename(dst)
        print(f"\n已执行 {len(renames)} 个重命名")
    else:
        print(f"\n[dry-run] 未实际执行，加 --execute 确认执行")

if __name__ == "__main__":
    main()
```

**改造点**：
- 排序规则：默认按修改时间排序，可改为按文件名`key=lambda p: p.name`或按大小。
- 命名模板：`name_parts`的拼接逻辑可自定义，如加入日期`datetime.now().strftime("%Y%m%d")`。
- 支持递归：将`directory.glob`改为`directory.rglob`可递归子目录。

**常见坑**：永远先dry-run再执行；目标文件名冲突会导致覆盖丢失，必须检查；Windows下文件名不区分大小写，`File.TXT`和`file.txt`视为同一文件；重命名涉及跨盘符时`rename`会失败，需用`shutil.move`。

## 2. 代码生成器：从配置/IDL生成C++结构体与序列化代码

**场景**：C++项目中消息结构体数量多且频繁变更，手写序列化/反序列化代码重复且易错。脚本从YAML/JSON配置生成C++头文件（结构体+序列化+size计算+打印），一次定义多处生成。

```python
#!/usr/bin/env python3
"""代码生成器：从YAML配置生成C++结构体与序列化代码。"""
import sys
import json
from pathlib import Path
from dataclasses import dataclass

# ============ 改造点：类型映射 ============
TYPE_MAP = {
    "uint8":  ("uint8_t",  "B", 1),
    "uint16": ("uint16_t", "H", 2),
    "uint32": ("uint32_t", "I", 4),
    "uint64": ("uint64_t", "Q", 8),
    "int8":   ("int8_t",   "b", 1),
    "int16":  ("int16_t",  "h", 2),
    "int32":  ("int32_t",  "i", 4),
    "int64":  ("int64_t",  "q", 8),
    "float":  ("float",    "f", 4),
    "double": ("double",   "d", 8),
    "bool":   ("bool",     "?", 1),
}

@dataclass
class FieldDef:
    name: str
    type: str
    array_size: int = 0  # 0表示标量

def parse_config(config: dict) -> list[tuple[str, list[FieldDef]]]:
    """解析配置，返回[(结构体名, [字段])]。"""
    structs = []
    for msg_name, fields in config.get("messages", {}).items():
        field_list = []
        for f in fields:
            ftype = f["type"]
            array_size = f.get("array", 0)
            field_list.append(FieldDef(f["name"], ftype, array_size))
        structs.append((msg_name, field_list))
    return structs

def gen_header(structs: list[tuple[str, list[FieldDef]]], guard: str) -> str:
    lines = []
    lines.append(f"#ifndef {guard}")
    lines.append(f"#define {guard}")
    lines.append("")
    lines.append("#include <cstdint>")
    lines.append("#include <cstring>")
    lines.append("#include <string>")
    lines.append("")
    lines.append("namespace protocol {")
    lines.append("")

    for struct_name, fields in structs:
        lines.append(f"struct {struct_name} {{")
        # 成员声明
        for f in fields:
            cpp_type = TYPE_MAP[f.type][0]
            if f.array_size > 0:
                lines.append(f"    {cpp_type} {f.name}[{f.array_size}];")
            else:
                lines.append(f"    {cpp_type} {f.name};")
        lines.append("")

        # 序列化大小
        lines.append("    static constexpr size_t fixed_size() {")
        total = sum(TYPE_MAP[f.type][2] * max(f.array_size, 1) for f in fields)
        lines.append(f"        return {total};")
        lines.append("    }")
        lines.append("")

        # 序列化（大端）
        lines.append("    size_t serialize(uint8_t* buf) const {")
        lines.append("        size_t offset = 0;")
        for f in fields:
            cpp_type, _, size = TYPE_MAP[f.type]
            if f.array_size > 0:
                lines.append(f"        for (int i = 0; i < {f.array_size}; ++i) {{")
                lines.append(f"            {cpp_type} v = {f.name}[i];")
                _emit_serialize_scalar(lines, f.type, "v", size)
                lines.append("        }")
            else:
                _emit_serialize_scalar(lines, f.type, f.name, size)
        lines.append("        return offset;")
        lines.append("    }")
        lines.append("")

        # 反序列化
        lines.append("    bool deserialize(const uint8_t* buf, size_t len) {")
        lines.append("        if (len < fixed_size()) return false;")
        lines.append("        size_t offset = 0;")
        for f in fields:
            cpp_type, _, size = TYPE_MAP[f.type]
            if f.array_size > 0:
                lines.append(f"        for (int i = 0; i < {f.array_size}; ++i) {{")
                _emit_deserialize_scalar(lines, f.type, f"{f.name}[i]", size)
                lines.append("        }")
            else:
                _emit_deserialize_scalar(lines, f.type, f.name, size)
        lines.append("        return true;")
        lines.append("    }")
        lines.append("};")
        lines.append("")

    lines.append("} // namespace protocol")
    lines.append("")
    lines.append(f"#endif // {guard}")
    return "\n".join(lines)

def _emit_serialize_scalar(lines: list[str], ftype: str, var: str, size: int):
    if size == 1:
        lines.append(f"        buf[offset++] = static_cast<uint8_t>({var});")
    else:
        # 大端：逐字节移位
        for i in range(size - 1, -1, -1):
            shift = i * 8
            lines.append(f"        buf[offset++] = static_cast<uint8_t>(({var} >> {shift}) & 0xFF);")

def _emit_deserialize_scalar(lines: list[str], ftype: str, var: str, size: int):
    if size == 1:
        lines.append(f"        {var} = static_cast<{TYPE_MAP[ftype][0]}>(buf[offset++]);")
    else:
        lines.append(f"        {var} = 0;")
        for i in range(size - 1, -1, -1):
            shift = i * 8
            lines.append(f"        {var} |= static_cast<{TYPE_MAP[ftype][0]}>(buf[offset++]) << {shift};")

def main():
    if len(sys.argv) < 2:
        print("用法: python3 codegen.py <config.json> [output.h]")
        sys.exit(1)

    config_path = Path(sys.argv[1])
    config = json.loads(config_path.read_text(encoding="utf-8"))
    structs = parse_config(config)

    guard = config.get("guard", config_path.stem.upper() + "_H")
    header = gen_header(structs, guard)

    out_path = Path(sys.argv[2]) if len(sys.argv) > 2 else Path("generated_messages.h")
    out_path.write_text(header, encoding="utf-8")
    print(f"已生成 {out_path}，包含 {len(structs)} 个结构体")

    # 打印配置示例
    example = {
        "guard": "PROTOCOL_MESSAGES_H",
        "messages": {
            "Heartbeat": [
                {"name": "version", "type": "uint8"},
                {"name": "seq", "type": "uint32"},
                {"name": "timestamp", "type": "uint64"},
            ],
            "DataFrame": [
                {"name": "frame_id", "type": "uint16"},
                {"name": "payload", "type": "uint8", "array": 64},
            ],
        },
    }
    print(f"\n配置示例: {json.dumps(example, indent=2, ensure_ascii=False)}")

if __name__ == "__main__":
    main()
```

**改造点**：
- `TYPE_MAP`：扩展支持的类型（如`string`变长字段需要特殊处理序列化逻辑）。
- 序列化字节序：当前生成大端代码，小端只需把移位顺序反过来或用`memcpy`+`htobe`宏。
- 配置格式：当前用JSON，可改为YAML（需`pip install pyyaml`）或自定义IDL格式。
- 生成内容：可扩展生成`operator==`、`toString()`、`nlohmann::json`转换等。

**常见坑**：生成代码的缩进必须精确，用列表拼接而非f-string嵌套更可控；结构体对齐问题——生成的序列化按紧凑布局，C++结构体本身可能有padding，`memcpy`直接序列化会出错，必须用逐字段序列化；`fixed_size`是编译期常量，适合栈上分配缓冲区；变长字段（string/vector）不能用`fixed_size`，需要单独处理。

## 4. 常见坑

- **生成前不清目标目录**：残留的旧文件会被误当成新生成物编译进来；生成前清空或按标记块替换。
- **生成物没有"自动生成"头**：一旦被人手改，下次生成即覆盖丢失；文件头写清来源与重新生成命令。
- **变长字段按定长处理**：`string`/`vector` 必须单独处理，`fixed_size` 仅用于定长类型，否则结构体错位。
- **批量重命名不做冲突检测**：先干跑打印计划、检测重名；Windows 文件系统不区分大小写要额外注意。
- **生成后不做编译校验**：生成的代码必须过一遍编译器，语法错误不能流入工程。
- **脚本非幂等**：重复运行会叠加副作用（重复插入、重复改名）；设计成可反复执行。
- **模板与数据没有分离**：改一个字段要动多处；把数据（CSV/JSON/YAML）与模板分开渲染。
